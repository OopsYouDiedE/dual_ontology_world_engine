using System.Reflection;
using Godot;
using DualOntologyWorldEngine.Gameplay;
using DualOntologyWorldEngine.Story;
public partial class GroundWireProbe : Node
{
 public GameEngine MakeEngine(){
 var game=new GameSession("{}","neutral","neutral");
 typeof(GameSession).GetField("_player",BindingFlags.NonPublic|BindingFlags.Instance)!.SetValue(game,new PlayerCandidateSession(_=>[],"A",100,100));
 var engine=new GameEngine();
 typeof(GameEngine).GetField("_session",BindingFlags.NonPublic|BindingFlags.Instance)!.SetValue(engine,game);
 return engine;
 }
}
