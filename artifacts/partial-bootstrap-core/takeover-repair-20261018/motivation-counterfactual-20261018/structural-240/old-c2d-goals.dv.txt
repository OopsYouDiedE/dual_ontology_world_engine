fn old_c2d_goals(actor):
    var goals = []
    let rows = c2d_rows()
    let tasks = c2daily_tasks()
    var t = 0
    while t < rt_count(tasks):
        if tasks[t].actor == actor.id:
            rt_push(goals, dv_with_many(c2_goal(dv_fact(c2daily_ref(tasks[t].field), true), tasks[t].when, [c2_approved("real-daily-contact-action")]), {"navigation_room": tasks[t].room, "navigation_methods": [tasks[t].id]}))
        t = t + 1
    rt_push(goals, dv_with_many(c2_observation_goal(c2daily_ref("meal_prepared"), c2_window(2, 690, 810), [c2_approved("look-for-real-lunch")], 2130), {"navigation_room": "living_hall", "navigation_methods": [rt_cat(actor.id, ".c2.inspect.c2.daily")]}))
    if actor.id == "ward":
        rt_push(goals, dv_with_many(c2_observation_goal(c2daily_ref("draft_notes"), c2daily_notes_window(), [c2_approved("inspect-current-draft-before-owned-norm-discussion")], 2250), {"navigation_room": "central_hall", "navigation_methods": ["ward.c2.daily.inspect-central"]}))
    if actor.id == "shen":
        rt_push(goals, dv_with_many(c2_goal(dv_fact(c2daily_ref("meal_cooked"), true), c2_window(2, 690, 810), [c2_approved("cook-real-lunch")]), {"navigation_room": "logistics", "navigation_methods": ["shen.c2.daily.cook-sixteen"]}))
        rt_push(goals, dv_with_many(c2_goal(dv_fact(c2daily_ref("meal_prepared"), true), dv_all([c2_window(2, 690, 810), dv_fact(c2daily_ref("meal_cooked"), true)]), [c2_approved("deliver-real-lunch")]), {"navigation_room": "living_hall", "navigation_methods": ["shen.c2.daily.deliver-sixteen"]}))
    if actor.id == "ruan":
        rt_push(goals, dv_with_many(c2_goal(dv_fact(c2daily_ref("draft_notes"), true), c2daily_notes_window(), [c2_approved("draft-notes-through-real-action")]), {"navigation_room": "central_hall", "navigation_methods": ["ruan.c2.draft-notes"]}))
    // Invocation-local reuse: actor/mind remain unchanged throughout this loop.
    // Evaluate lazily at the original first use so unrelated errors keep order.
    var concerned_ready = false
    var concerned = false
    var notice_ready = false
    var notice = false
    var case_ready = false
    var case_gate = false
    var i = 0
    while i < rt_count(rows):
        let row = rows[i]
        if row.actor == actor.id and (not dv_find(row, "manual_only", false) or (dv_has_key(row, "manipulation") and dv_find(actor.policy, "allow_manipulation_candidates", false))):
            var when = dv_all([row.when, row.literal])
            if not dv_find(row, "repeatable", false):
                when = dv_all([when, dv_fact(c2d_mark(actor.id, row.id), false)])
            if dv_has_key(row, "owned_decision"):
                let candidate = dvp_find_action(actor.actions, row.id)
                if candidate == false or not dvp_owned_decision_ready(actor, candidate, actor.mind):
                    when = dv_not(dv_true())
            if dv_has_key(row, "partner_decision"):
                let candidate = dvp_find_action(actor.actions, row.id)
                if candidate == false or not dvp_personal_ready(actor.mind, candidate.personal_choice):
                    when = dv_not(dv_true())
            if dv_find(row, "work_refusal", false) or dv_find(row, "concern_question", false):
                if not concerned_ready:
                    concerned = c2_quer_concerned(actor)
                    concerned_ready = true
                if not concerned:
                    when = dv_not(dv_true())
            if dv_find(row, "safety_warning", false) or dv_find(row, "guard_decision", false):
                if not notice_ready:
                    notice = c2_owned_safety_notice(actor)
                    notice_ready = true
                if not notice:
                    when = dv_not(dv_true())
            if dv_find(row, "category", "") == "case":
                if not case_ready:
                    case_gate = c2d_case_gate(actor.mind)
                    case_ready = true
                when = dv_all([when, case_gate])
            if dv_has_key(row, "received_speech_requirement"):
                if not c2_received_requirement(actor, row.received_speech_requirement):
                    when = dv_not(dv_true())
            if dv_has_key(row, "response_topic"):
                if rt_count(dvs_pending_request(actor, row)) == 0:
                    when = dv_not(dv_true())
            if dv_has_key(row, "advice_decision"):
                let candidate = dvp_find_action(actor.actions, row.id)
                if candidate == false or not dvs_advice_decision_ready(actor, candidate, actor.mind):
                    when = dv_not(dv_true())
            rt_push(goals, c1n_index_goal_window({"kind": "communicate", "scope": "personal", "method_id": row.id, "available_when": when, "basis": [row.source], "requires_argument": false, "navigation_room": row.room}))
        i = i + 1
    let study = dv_all([dv_fact(c2_access("north"), true), c2_window(2, 600, 690)])
    rt_push(goals, dv_with_many(c2_goal(c2_at(actor.id, "lecture"), study, [c2d_source("06_场面.story", 79, "请到讲堂")]), {"navigation_room": "lecture", "maintain_presence": true}))
    if actor.id == "bertrand":
        rt_push(goals, dv_with(c2_goal(dv_fact(c2d_person_ref(actor.id, "watching"), true), c2_any_night(), [c1_source("godot_csharp/scenes/Awakening/Script/story/生态/人物/14_贝尔特朗.story", 30, "夜里巡夜：护住同伴")]), "navigation_room", "logistics"))
        rt_push(goals, dv_with(c2_observation_goal(c2d_person_ref(actor.id, "small_shadow_seen"), dv_all([c2_any_night(), dv_fact(c2d_person_ref(actor.id, "watching"), true)]), [c2d_source("04_案件.story", 25, "有限身影观察")], 0), "navigation_room", "logistics"))
    if actor.id == "detective":
        let ledger_fields = ["counted", "reserve_received", "day_recorded", "night_recorded", "pending_registered"]
        rt_push(goals, c2_observation_goal(c2_ref("c2.counter", "remaining"), c2_at(actor.id, "logistics"), [c1_source("docs/design-standard.md", 0, "实物清点与台账对照")], 0))
        i = 0
        while i < rt_count(ledger_fields):
            rt_push(goals, c2_observation_goal(c2_ref("c2.ledger", ledger_fields[i]), c2_at(actor.id, "logistics"), [c2d_source("04_案件.story", 10, "从读到账目再判断是否有缺口")], 0))
            i = i + 1
    if actor.id == "quer":
        rt_push(goals, dv_with(c2_goal(dv_fact(c2_ref("c2.granary", "latch_loose"), false), dv_all([dv_fact(c2_ref("c2.granary", "latch_loose"), true), c2_work_accepted(actor.id, "quer.c2.granary.precision")]), [c2d_source("10_执行.story", 17, "RepairLatch")]), "navigation_room", "logistics"))
    let opportunity = c2d_known_council_proposal()
    let venues = [{"room": "oath_court", "wind": true}, {"room": "central_hall", "wind": false}]
    i = 0
    while i < rt_count(venues):
        let venue = venues[i]
        let when = dv_all([c2d_council_when(), c2_alive(actor.id), opportunity, dv_fact(c2_ref("c2.weather", "wind"), venue.wind), dv_fact(c2daily_ref(rt_cat("signed.", actor.id)), false)])
        rt_push(goals, dv_with(c2_goal(c2_at(actor.id, venue.room), when, [c2d_source("06_场面.story", 340, "章二议事机会")]), "navigation_room", venue.room))
        // Ballot effects increment the actually inspected tally, never an assumed zero.
        let tally = ["votes.open_plan", "votes.seal_plan", "votes.committee_plan"]
        var n = 0
        while n < rt_count(tally):
            rt_push(goals, dv_with(c2_observation_goal(c2daily_ref(tally[n]), when, [c2_approved("actual-council-tally-before-personal-ballot")], 6210), "navigation_room", venue.room))
            n = n + 1
        i = i + 1
    return goals
